a. Simulate solution for 4-Queen / N-Queen problem
def safe(b, r, c):
    for i in range(len(b)):
        if b[i] == c or abs(b[i]-c) == abs(i-r):
            return False
    return True

def solve(n, b=[]):
    if len(b) == n:
        for r in range(n):
            print(" ".join("Q" if c == b[r] else "." for c in range(n)))
        print()
        return 1

    total = 0
    for c in range(n):
        if safe(b, len(b), c):
            total += solve(n, b+[c])
    return total

n = int(input("Enter N: "))
print("Solutions:", solve(n))



b. Constraint satisfaction problem: Map Coloring

graph = {'A':['B','C'],'B':['A','C','D'],'C':['A','B','D'],'D':['B','C']}
colors = ['Red','Green','Blue']
result = {}

for node in graph:
    for color in colors:
        if all(result.get(n) != color for n in graph[node]):
            result[node] = color
            break

for node in result:
    print(node, "->", result[node])
