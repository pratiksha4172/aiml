8. Write a program which contains three predicates: male, female, 
parent. Make rules for following family relations: father, mother, 
grandfather, grandmother, brother, sister, uncle, aunt, nephew and 
niece, cousin.
Question:
a. Define: Clauses, Facts, Predicates and Rules with conjunction and 
disjunction b. Draw Family Tree.


male(john).
male(bob).
male(sam).
female(mary).
female(lisa).

parent(john, bob).
parent(mary, bob).
parent(bob, sam).
parent(lisa, sam).

father(X,Y) :-
    male(X),
    parent(X,Y).

mother(X,Y) :-
    female(X),
    parent(X,Y).

grandfather(X,Y) :-
    male(X),
    parent(X,Z),
    parent(Z,Y).

grandmother(X,Y) :-
    female(X),
    parent(X,Z),
    parent(Z,Y).

brother(X,Y) :-
    male(X),
    parent(Z,X),
    parent(Z,Y),
    X \= Y.

sister(X,Y) :-
    female(X),
    parent(Z,X),
    parent(Z,Y),
    X \= Y.
