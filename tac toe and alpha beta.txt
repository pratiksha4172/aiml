a. Simulation of tic – tac – toe game using Min-Max Search

board = [" "] * 9

def win():
    lines = [(0,1,2),(3,4,5),(6,7,8),
             (0,3,6),(1,4,7),(2,5,8),
             (0,4,8),(2,4,6)]

    for a,b,c in lines:
        if board[a] == board[b] == board[c] != " ":
            return board[a]
    if " " not in board:
        return "Draw"

def minimax(turn):
    r = win()

    if r == "O": return 1
    if r == "X": return -1
    if r == "Draw": return 0

    scores = []

    for i in range(9):
        if board[i] == " ":
            board[i] = "O" if turn else "X"
            scores.append(minimax(not turn))
            board[i] = " "

    return max(scores) if turn else min(scores)

def best_move():
    best = -1
    move = 0

    for i in range(9):
        if board[i] == " ":
            board[i] = "O"
            score = minimax(False)
            board[i] = " "

            if score > best:
                best = score
                move = i

    return move

print("Best Move:", best_move())
print("Position:", best_move() + 1)


b. Alpha Beta Pruning

def ab(d,i,mx,v,a,b):
    if d==3: return v[i]
    if mx:
        best=-1000
        for j in range(2):
            best=max(best,ab(d+1,i*2+j,False,v,a,b)); a=max(a,best)
            if b<=a: break
        return best
    else:
        best=1000
        for j in range(2):
            best=min(best,ab(d+1,i*2+j,True,v,a,b)); b=min(b,best)
            if b<=a: break
        return best

print(ab(0,0,True,[3,5,6,9,1,2,0,-1],-1000,1000))
